cmake_minimum_required(VERSION 3.25)
project(magma VERSION 0.7.3 DESCRIPTION "Windows, input and rendering for Basalt applications"
  LANGUAGES CXX)

option(MAGMA_BUILD_TESTS "Build the Catch2 test suite" ${PROJECT_IS_TOP_LEVEL})
option(MAGMA_BUILD_BENCHMARKS "Build the Catch2 benchmarks" OFF)
option(MAGMA_BUILD_EXAMPLES "Build the example programs" ${PROJECT_IS_TOP_LEVEL})
option(MAGMA_INSTALL "Add install and package export rules" ${PROJECT_IS_TOP_LEVEL})
option(MAGMA_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ${PROJECT_IS_TOP_LEVEL})
option(MAGMA_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
# The Metal backend is written in Objective-C++, which only Clang compiles.
if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set(_magma_metal_default ON)
else()
  set(_magma_metal_default OFF)
endif()
option(MAGMA_METAL "Build the Metal backend (macOS with Clang)" ${_magma_metal_default})
if(MAGMA_METAL)
  enable_language(OBJCXX)
endif()
# The Vulkan backend loads the Vulkan library when it starts, so building it needs no Vulkan SDK.
# Vulkan on macOS would need MoltenVK, and Metal replaces it there.
if(NOT APPLE)
  set(_magma_vulkan_default ON)
else()
  set(_magma_vulkan_default OFF)
endif()
option(MAGMA_VULKAN "Build the Vulkan backend" ${_magma_vulkan_default})
if(WIN32)
  set(_magma_d3d12_default ON)
else()
  set(_magma_d3d12_default OFF)
endif()
option(MAGMA_D3D12 "Build the Direct3D 12 backend (Windows only)" ${_magma_d3d12_default})
if(MAGMA_D3D12 AND NOT WIN32)
  message(FATAL_ERROR "MAGMA_D3D12 needs Windows.")
endif()
if(MAGMA_VULKAN)
  # volk is a C file.
  enable_language(C)
endif()
set(MAGMA_SDL "bundled" CACHE STRING
  "Where SDL 3 comes from: bundled (built from the pinned release) or system (find_package)")
set_property(CACHE MAGMA_SDL PROPERTY STRINGS bundled system)
if(NOT MAGMA_SDL MATCHES "^(bundled|system)$")
  message(FATAL_ERROR "MAGMA_SDL must be 'bundled' or 'system', not '${MAGMA_SDL}'.")
endif()

if(PROJECT_IS_TOP_LEVEL)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  set(CMAKE_DEBUG_POSTFIX d)
endif()

include(GNUInstallDirs)
include("${PROJECT_SOURCE_DIR}/cmake/compiler_options.cmake")
include("${PROJECT_SOURCE_DIR}/cmake/dependencies.cmake")

add_subdirectory(src)

if(MAGMA_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(MAGMA_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()

if(MAGMA_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()

if(MAGMA_INSTALL)
  include("${PROJECT_SOURCE_DIR}/cmake/install.cmake")
endif()
